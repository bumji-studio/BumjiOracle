# ทฤษฎีที่ 21: 六亲 “หลักชิง” วงศ์ญาติที่ให้โทษ — “ศัตรูนับร้อยพัน ไม่น่ากลัวเท่าคนใกล้ตัวหักหลัง”

> **แหล่งที่มา:** เพจ *แม่ไม้ดวงจีน 八字 : Destiny Detective* (โพสต์ทฤษฎี “คนใกล้ตัวหักหลัง”)  
> **หมวดหมู่:** ทฤษฎีวงศ์ญาติ (六亲) / การระบุศัตรูและผู้ทรยศ / การอ่านสิบเทพที่เป็น 忌神 (ธาตุให้โทษ)  

---

## 1. ปรัชญาการระบุ “คนใกล้ตัว” ในดวงชะตา

> *“ศัตรูนับร้อยพัน ยังไม่น่ากลัวเท่า คนใกล้ตัวหักหลัง...  
> คำว่า ‘คนใกล้ตัว’ นั้นกว้างมาก แต่ 八字 ‘โป๊ยหยี่’ สามารถระบุได้อย่างแม่นยำว่าเป็นผู้ใด ผ่าน 六亲 ‘หลักชิง’ วงศ์ญาติ!”*

ในวิชาดวงจีน เมื่อดาวสิบเทพตัวใดตัวหนึ่งกลายเป็น **忌神 “กี่ซิ้ง” (ธาตุที่ให้โทษกับดวงชะตา)**  
บุคคลที่เป็นตัวแทนของสิบเทพตัวนั้น จะแปรเปลี่ยนบทบาทจากคนคุ้นเคย กลายมาเป็น **“ผู้สร้างความเดือดร้อน”**:
* คอยให้ร้าย ใส่ร้ายป้ายสี
* เข้ามากดดัน จู้จี้จุกจิก ก้าวก่ายสร้างความวุ่นวาย
* เกิดความอิจฉาริษยา
* วางแผนหักหลัง ทรยศ แทงข้างหลัง
* เบียดเบียน โกงทรัพย์สิน หรือแย่งชิงผลประโยชน์ (แย่งลาภ)

---

## 3. หลักการพิจารณาเชิงลึก: “ดูดีดี คนใกล้ตัวคือผู้ใด”

1. **ระบุดิถีและตรวจความอ่อน-แข็งของดวง:**
   * เพื่อจำแนกให้ขาดว่าธาตุใดคือ **用神 (ให้คุณ)** และธาตุใดคือ **忌神 (ให้โทษ)**
2. **ตรวจดูธาตุที่จรเข้ามา (ปีจร / วัยจร):**
   * หากปีจรนำพาธาตุที่เป็น **忌神** เข้ามา ให้ดูว่าสัมพันธ์เป็นสิบเทพตัวใด
   * และตรวจสอบว่าไปกระทบ (ชง, เฮ้ง, ผั่วะ, ไห่) กับเสาใดในดวงเดิม
3. **ตัวอย่างการประเมินภัยจากคนใกล้ตัว:**
   * *หญิงดิถีทอง 辛 (ดวงแข็ง ไม่เอาทองและดิน):* เมื่อปีจรเป็นธาตุดิน (印 อิ่ง) เข้ามา $\rightarrow$ ระวังแม่ ผู้ใหญ่ หรือเจ้านาย เข้ามากดดันสร้างความวุ่นวายจนหน้าที่การงานเสียหาย
   * *ชายดิถีไม้ 甲 (ดวงอ่อน ไม่เอาทองและไฟ/ดิน):* เมื่อปีจรเป็นธาตุทอง (官/殺 กัวสั่วะ) เข้ามา $\rightarrow$ ระวังลูกน้องหรือบริวารทรยศ แทงข้างหลัง ขโมยข้อมูล นำพาคดีความมาสู่ตนเอง

## 4. แนวทางการนำไปเขียนฟังก์ชันวิเคราะห์ในโปรแกรม (`bazi.js`)

```javascript
// ฟังก์ชันวิเคราะห์บุคคลที่อาจให้โทษ/สร้างความเดือดร้อน (Toxic Kinship Analysis)
function identifyBetrayerRisk(dayMasterStem, gender, unfavorableGods) {
  const risks = [];

  unfavorableGods.forEach(god => {
    switch (god) {
      case 'Friend':
      case 'Rob_Wealth':
        risks.push({
          group: '比劫 (ปี่เกี๊ยบ)',
          people: 'พี่น้อง เพื่อนสนิท หุ้นส่วนทางธุรกิจ หรือเพื่อนร่วมงาน',
          warning: 'ระวังการหักหลังเรื่องเงินทอง การแย่งลูกค้า แย่งผลประโยชน์ หรือถูกโกง'
        });
        break;
      case 'Eating_God':
      case 'Hurting_Officer':
        if (gender === 'female') {
          risks.push({
            group: '食傷 (เจี้ยะเซีย)',
            people: 'บุตร บริวาร ลูกน้อง ลูกจ้าง หรือคนในความดูแล',
            warning: 'ระวังบริวารแทงข้างหลัง ลูกน้องทำงานผิดพลาดสร้างความเสียหาย หรือลูกหลานสร้างปัญหา'
          });
        }
        break;
      case 'Direct_Wealth':
      case 'Indirect_Wealth':
        risks.push({
          group: '財 (ไช้)',
          people: gender === 'male' ? 'ภรรยา คนรัก หรือบิดา' : 'บิดา หรือบุตรหลาน',
          warning: 'ระวังถูกคนรักหรือคนในครอบครัวเบียดเบียนเงินทอง นำพาหนี้สินมาให้'
        });
        break;
      case 'Direct_Officer':
      case 'Seven_Killings':
        if (gender === 'female') {
          risks.push({
            group: '官殺 (กัวสั่วะ)',
            people: 'สามี หรือคนรัก',
            warning: 'ระวังถูกสามีกดขี่ ข่มเหง นอกใจ นำเรื่องเดือดร้อนหรือปัญหาคดีความมาสู่'
          });
        } else {
          risks.push({
            group: '官殺 (กัวสั่วะ)',
            people: 'ลูกน้อง บริวาร หรือคนใต้บังคับบัญชา',
            warning: 'ระวังลูกน้องก่อเรื่องผิดกฎหมาย นำพาคดีความ หรือแทงข้างหลัง'
          });
        }
        break;
      case 'Direct_Resource':
      case 'Indirect_Resource':
        risks.push({
          group: '印 (อิ่ง)',
          people: 'มารดา ผู้ใหญ่ เจ้านาย หรือผู้มีอำนาจปกครอง',
          warning: 'ระวังผู้ใหญ่เข้ามากดดัน ควบคุม ก้าวก่ายชีวิต หรือเจ้านายกลั่นแกล้งใส่ร้าย'
        });
        break;
    }
  });

  return risks;
}
```
